cmake_minimum_required(VERSION 3.16)
project(obca_navigation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(PkgConfig REQUIRED)
pkg_check_modules(IPOPT REQUIRED IMPORTED_TARGET ipopt)
add_library(obca_core src/core.cpp src/solver.cpp)
target_include_directories(obca_core PUBLIC include)
target_link_libraries(obca_core PUBLIC PkgConfig::IPOPT)

option(OBCA_STANDALONE "Build and test the ROS-independent core only" OFF)
if(NOT OBCA_STANDALONE)
  find_package(ament_cmake REQUIRED)
  set(dependencies rclcpp f110_msgs ackermann_msgs geometry_msgs nav_msgs sensor_msgs
                   std_msgs diagnostic_msgs tf2 tf2_ros)
  foreach(dependency IN LISTS dependencies)
    find_package(${dependency} REQUIRED)
  endforeach()
  add_executable(planner_node src/planner_node.cpp)
  add_executable(tracker_node src/tracker_node.cpp)
  foreach(node planner_node tracker_node)
    target_link_libraries(${node} obca_core)
    ament_target_dependencies(${node} ${dependencies})
  endforeach()
  install(TARGETS planner_node tracker_node RUNTIME DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config launch docs DESTINATION share/${PROJECT_NAME})
endif()
if(BUILD_TESTING OR OBCA_STANDALONE)
  enable_testing()
  target_compile_definitions(obca_core PRIVATE OBCA_TESTING)
  add_executable(core_test test/core_test.cpp)
  target_link_libraries(core_test obca_core)
  add_test(NAME obca_core_checks COMMAND core_test)
endif()
if(NOT OBCA_STANDALONE)
  ament_package()
endif()
